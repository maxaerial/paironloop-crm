CREATE OR REPLACE FUNCTION public.handle_new_user() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE new_company_id uuid; company_name text; v_vorname text; v_nachname text; BEGIN company_name := NEW.raw_user_meta_data->>'company_name'; v_vorname := NEW.raw_user_meta_data->>'vorname'; v_nachname := NEW.raw_user_meta_data->>'nachname'; IF company_name IS NULL OR company_name = '' THEN company_name := split_part(NEW.email, '@', 1); END IF; INSERT INTO public.companies (name, plan) VALUES (company_name, 'free') RETURNING id INTO new_company_id; INSERT INTO public.profiles (id, company_id, role, vorname, nachname) VALUES (NEW.id, new_company_id, 'admin', v_vorname, v_nachname) ON CONFLICT (id) DO UPDATE SET company_id = new_company_id, role = 'admin', vorname = v_vorname, nachname = v_nachname; RETURN NEW; END; $$;
